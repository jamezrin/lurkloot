# Site Automatic Localization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Localize the Astro marketing site and changelog for all eleven extension locales from a single English source, generating static `/es/`-style routes at deploy time with Llama 4 Scout.

**Architecture:** English copy stays in `copy/en.ts` and `changelog.json`. A hash-addressed cache plus a `Translator` port fill other locales. Astro renders shared views once per locale. Workers AI runs only in the trusted site-deploy build, never during `pnpm check`.

**Tech Stack:** Astro 7 static site, TypeScript, Node `node:test`, Cloudflare Workers AI REST (`@cf/meta/llama-4-scout-17b-16e-instruct`), GitHub Actions cache, `@lurkloot/shared` locales.

## Global Constraints

- English is the only authored source; do not add site strings to `packages/locales`.
- Locales are exactly `SupportedLocale`: `en es fr it ru de zh_CN hi pt_BR ar tr`.
- URL prefixes: English unprefixed; `zh_CN` → `zh-cn`; `pt_BR` → `pt-br`; others lowercase as-is.
- Privacy policy body and `/privacy` stay English; do not emit `/es/privacy`.
- Translated pages are `noindex`; English URLs remain canonical; sitemap lists English only.
- `astro build` must not call the network. Translation is `pnpm --filter @lurkloot/site translate`.
- Model: `@cf/meta/llama-4-scout-17b-16e-instruct`. Set `max_tokens` to at least `8192`.
- `WORKERS_AI_API_TOKEN` is Workers AI Edit only — never reuse `CLOUDFLARE_API_TOKEN`.
- PR / `pnpm check` stay credential-free; missing cache entries render English.
- Preserve two-space indent, double quotes, semicolons, `type` imports.
- Diagnostic/extension catalog rules are unchanged. Site FAQ JSON-LD only on English home.

```ts
import type { SupportedLocale } from "@lurkloot/shared/models";
import { SUPPORTED_LOCALES } from "@lurkloot/shared/settings";
import { isRtlLocale } from "@lurkloot/shared/i18n";

export const SITE_LOCALES: readonly SupportedLocale[] = SUPPORTED_LOCALES;

export type SitePath = "/" | "/changelog";

export function localeToPrefix(locale: SupportedLocale): string {
  if (locale === "en") return "";
  return locale.replaceAll("_", "-").toLowerCase();
}

export function prefixToLocale(prefix: string): SupportedLocale | undefined {
  const normalized = prefix.replace(/\/+$/, "").replace(/^\/+/, "").toLowerCase();
  if (normalized === "") return "en";
  return SITE_LOCALES.find((locale) => localeToPrefix(locale) === normalized);
}

export function pageHref(locale: SupportedLocale, path: SitePath): string {
  const prefix = localeToPrefix(locale);
  if (!prefix) return path;
  return path === "/" ? `/${prefix}/` : `/${prefix}${path}`;
}

export function parseLocaleFromPathname(pathname: string): { locale: SupportedLocale; path: SitePath | string } {
  const parts = pathname.split("/").filter(Boolean);
  const maybe = parts[0] ? prefixToLocale(parts[0]) : "en";
  if (maybe && maybe !== "en") {
    const rest = `/${parts.slice(1).join("/")}`.replace(/\/$/, "") || "/";
    return { locale: maybe, path: rest };
  }
  const path = `/${parts.join("/")}`.replace(/\/$/, "") || "/";
  return { locale: "en", path };
}

export function htmlLang(locale: SupportedLocale): string {
  return locale.replace("_", "-");
}

export function htmlDir(locale: SupportedLocale): "ltr" | "rtl" {
  return isRtlLocale(locale) ? "rtl" : "ltr";
}

export function prefixedLocales(): Exclude<SupportedLocale, "en">[] {
  return SITE_LOCALES.filter((locale): locale is Exclude<SupportedLocale, "en"> => locale !== "en");
}
```

`packages/site/src/i18n/strings.ts`:

```ts
import { hashEnglish } from "./hash.ts";

export function collectStringLeaves(value: unknown, out: string[] = []): string[] {
  if (typeof value === "string") {
    out.push(value);
    return out;
  }
  if (Array.isArray(value)) {
    for (const item of value) collectStringLeaves(item, out);
    return out;
  }
  if (value && typeof value === "object") {
    for (const nested of Object.values(value)) collectStringLeaves(nested, out);
  }
  return out;
}

export function applyStringLeaves<T>(value: T, leaves: string[], index = { i: 0 }): T {
  if (typeof value === "string") {
    const next = leaves[index.i];
    if (next === undefined) throw new Error("Translation leaf count is shorter than the English tree");
    index.i += 1;
    return next as T;
  }
  if (Array.isArray(value)) {
    return value.map((item) => applyStringLeaves(item, leaves, index)) as T;
  }
  if (value && typeof value === "object") {
    const entries = Object.entries(value as Record<string, unknown>).map(([key, nested]) => [
      key,
      applyStringLeaves(nested, leaves, index),
    ]);
    return Object.fromEntries(entries) as T;
  }
  return value;
}

export function translationId(text: string): string {
  return hashEnglish(text);
}
```

`packages/site/src/i18n/cache.ts`:

```ts
import { mkdir, readFile, writeFile } from "node:fs/promises";
import { join } from "node:path";
import type { SupportedLocale } from "@lurkloot/shared/models";

export function defaultCacheDir(from = import.meta.dirname): string {
  return join(from, "../../.i18n-cache");
}

export function cacheFile(dir: string, locale: Exclude<SupportedLocale, "en">): string {
  return join(dir, `${locale}.json`);
}

export async function readCache(
  dir: string,
  locale: Exclude<SupportedLocale, "en">,
): Promise<Record<string, string>> {
  try {
    const raw = await readFile(cacheFile(dir, locale), "utf8");
    const parsed = JSON.parse(raw) as unknown;
    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return {};
    return Object.fromEntries(
      Object.entries(parsed).filter((entry): entry is [string, string] => typeof entry[1] === "string"),
    );
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === "ENOENT") return {};
    throw error;
  }
}

export async function writeCache(
  dir: string,
  locale: Exclude<SupportedLocale, "en">,
  entries: Record<string, string>,
): Promise<void> {
  await mkdir(dir, { recursive: true });
  const sorted = Object.fromEntries(Object.entries(entries).sort(([a], [b]) => a.localeCompare(b)));
  await writeFile(cacheFile(dir, locale), `${JSON.stringify(sorted, null, 2)}\n`);
}
```

```ts
import type { SupportedLocale } from "@lurkloot/shared/models";
import { readCache } from "./cache.ts";
import { translationId } from "./strings.ts";

export async function resolveLeaves(input: {
  cacheDir: string;
  locale: SupportedLocale;
  english: string[];
}): Promise<string[]> {
  if (input.locale === "en") return input.english;
  const cache = await readCache(input.cacheDir, input.locale);
  return input.english.map((text) => cache[translationId(text)] ?? text);
}
```

```ts
import { changelog } from "../changelog.ts";
import { englishCopy } from "../copy/en.ts";
import { applyStringLeaves, collectStringLeaves } from "./strings.ts";
import { resolveLeaves } from "./resolve.ts";
import { defaultCacheDir } from "./cache.ts";
import type { SupportedLocale } from "@lurkloot/shared/models";
import type { SiteCopy } from "../copy/types.ts";
import type { ChangelogEntry } from "../changelog.ts";

export function collectSiteEnglish(): string[] {
  return [
    ...collectStringLeaves(englishCopy),
    ...changelog.flatMap((entry) => entry.changes.map((change) => change.text)),
  ];
}

export async function loadSiteCopy(locale: SupportedLocale, cacheDir = defaultCacheDir()): Promise<SiteCopy> {
  const english = collectStringLeaves(englishCopy);
  const resolved = await resolveLeaves({ cacheDir, locale, english });
  return applyStringLeaves(englishCopy, resolved);
}

export async function loadChangelog(locale: SupportedLocale, cacheDir = defaultCacheDir()): Promise<ChangelogEntry[]> {
  const english = changelog.flatMap((entry) => entry.changes.map((change) => change.text));
  const resolved = await resolveLeaves({ cacheDir, locale, english });
  let i = 0;
  return changelog.map((entry) => ({
    ...entry,
    changes: entry.changes.map((change) => ({ ...change, text: resolved[i++] ?? change.text })),
  }));
}
```
